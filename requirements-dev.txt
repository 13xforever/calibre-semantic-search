# NOT shipped: the plugin ZIP contains only src/ and runs on calibre's Python with

lancedb==0.38.0  # hard test dependency - the lancedb backend tests must run, not skip

                 # lxml version. The dev gate re-runs the chunker tests under calibre-debug (CALIBRE_DEBUG in .env, or PATH) to cover
                 # calibre's parser; that run is skipped with a note when unavailable.
numpy==2.5.3     # optional at runtime (pure-Python fallback), but needed here to exercise the numpy paths in tests; calibre does not bundle it
zstandard==0.25.0  # zstd text codec; without it a fresh store falls back to zlib
